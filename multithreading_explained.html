    def get_empty_kvs(self, empty_keys, empty_secrets):
        empty_secrets_ids = {e['keyvault_id'] for e in empty_secrets}
        fully_empty = []
        for ek in empty_keys:
            if ek['keyvault_id'] in empty_secrets_ids:
                fully_empty.append({
                    'keyvault_id': ek['keyvault_id'],
                    'keyvault_name': ek['keyvault_name'],
                    'resource_group': ek['resource_group'],
                    'subscription_id': ek['subscription_id'],
                    'tenant': ek['tenant'],
                    'cloud': ek['cloud']
                })
        return fully_empty

    def get_all_kvs(self,cloud):
        argClient = ResourceGraph().get_resorce_graph(cloud)
        argQueryOptions = arg.models.QueryRequestOptions(result_format="objectArray", top=1000)
        strQuery = """
        resources
        | where type == "microsoft.keyvault/vaults"
        | summarize ids = make_list(id, 10000) by resourceGroup,subscriptionId,tenantId
        | extend bag = pack('resource_group',resourceGroup,'id',ids)
        | summarize keyvaults = make_list(bag, 10000) by subscriptionId,tenantId
        """
        argQuery = arg.models.QueryRequest(query=strQuery, options=argQueryOptions)
        argResults = argClient.resources(argQuery)
        result = argResults.as_dict()
        while("skip_token" in result.keys() and result["skip_token"]!= None):
            argQueryOptions = arg.models.QueryRequestOptions(skip_token=result["skip_token"],result_format="objectArray", top=1000)
            argQuery = arg.models.QueryRequest(query=strQuery, options=argQueryOptions)
            argResults = argClient.resources(argQuery).as_dict()
            result["count"] += argResults["count"]
            result["data"] += argResults["data"]
            result["skip_token"] = argResults["skip_token"] if "skip_token" in argResults.keys() else None
        return result["data"]

    def secrets_paginate(self, kvs, cloud):
        secrets = []
        empty_secrets = []
        try:
            kv_client = KeyVaultClient().get_kv_client(kvs['subscriptionId'], cloud)
            tenant = kvs['tenantId']
        except Exception as e:
            logging.error(f"Cannot create KV client for subscription {kvs.get('subscriptionId')}: {e}")
            return [], []
        for kv in kvs['keyvaults']:
            for kv_id in kv['id']:
                logging.info(f"Processing Key Vault {kv_id}")
                vault_name = kv_id.split('/')[-1]
                try:
                    secrets_list = list(kv_client.secrets.list(kv['resource_group'], vault_name))

                    if not secrets_list:
                        empty_secrets.append({
                            'keyvault_id': kv_id,
                            'keyvault_name': vault_name,
                            'resource_group': kv['resource_group'],
                            'subscription_id': kvs['subscriptionId'],
                            'tenant': tenant,
                            'cloud': cloud,
                            'secrets_count': 0
                        })
                        continue

                    for secret in secrets_list:
                        temp = {}
                        temp['id'] = secret.id
                        temp['name'] = secret.name
                        temp['created_on'] = secret.properties.attributes.created.strftime("%m/%d/%Y %H:%M:%S") if secret.properties.attributes.created else None
                        temp['expires_on'] = secret.properties.attributes.expires.strftime("%m/%d/%Y %H:%M:%S") if secret.properties.attributes.expires else None
                        temp['content_type'] = secret.properties.content_type
                        temp['enabled'] = secret.properties.attributes.enabled
                        temp['updated'] = secret.properties.attributes.updated.strftime("%m/%d/%Y %H:%M:%S") if secret.properties.attributes.updated else None
                        temp['properties'] = secret.properties.as_dict()
                        temp['tags'] = secret.tags
                        temp['keyvault_name'] = vault_name
                        temp['resource_group'] = kv['resource_group']
                        temp['subscription_id'] = kvs['subscriptionId']
                        temp['tenant'] = tenant
                        secrets.append(temp)
                except Exception as e:
                    logging.error(f"Error processing Key Vault {vault_name}: {e}")
                    continue
                logging.info(f"Finished processing Key Vault {kv_id} current total secrets {len(secrets)}")
        return secrets, empty_secrets

    def keys_paginate(self, kvs, cloud):

        empty_keys = []

            kv_client = KeyVaultClient().get_kv_client(kvs['subscriptionId'], cloud)

                vault_name = kv_id.split('/')[-1]
                try:
                    keys_list = list(kv_client.keys.list(kv['resource_group'], vault_name))

                    if not keys_list:
                        empty_keys.append({
                            'keyvault_id': kv_id,
                            'keyvault_name': vault_name,
                            'resource_group': kv['resource_group'],
                            'subscription_id': kvs['subscriptionId'],
                            'tenant': tenant,
                            'cloud': cloud,
                            'keys_count': 0
                        })

                    for key in keys_list:
                        temp = {}
                        temp['id'] = key.id
                        temp['name'] = key.name
                        temp['created_on'] = datetime.fromtimestamp(key.attributes.created).strftime("%m/%d/%Y %H:%M:%S") if key.attributes.created else None
                        temp['expires_on'] = datetime.fromtimestamp(key.attributes.expires).strftime("%m/%d/%Y %H:%M:%S") if key.attributes.expires else None
                        temp['key_type'] = key.kty
                        temp['key_size'] = key.key_size
                        temp['enabled'] = key.attributes.enabled
                        temp['updated'] = datetime.fromtimestamp(key.attributes.updated).strftime("%m/%d/%Y %H:%M:%S") if key.attributes.updated else None
                        temp['attributes'] = key.attributes.as_dict()
                        temp['keyvault_name'] = vault_name
                        temp['resource_group'] = kv['resource_group']
                        temp['subscription_id'] = kvs['subscriptionId']
                        temp['tenant'] = tenant
                        keys.append(temp)
                except Exception as e:
                    logging.error(f"Error processing Key Vault {vault_name}: {e}")
                    continue
                logging.info(f"Finished processing Key Vault {kv_id} current total keys {len(keys)}")
        return keys, empty_keys

    def process_secrets(self, kv, cloud):
        return self.secrets_paginate(kv, cloud)

    def process_keys(self, kv, cloud):
        return self.keys_paginate(kv, cloud)

    def kv_secrets_api(self, cloud):
        secrets = []
        empty_secrets = []
        kvs = self.get_all_kvs(cloud)
        with concurrent.futures.ThreadPoolExecutor() as executor:
            results = list(executor.map(lambda args: self.process_secrets(*args), [(kv, cloud) for kv in kvs]))
            for s, es in results:
                secrets.extend(s)
                empty_secrets.extend(es)
        return secrets, empty_secrets

    def kv_keys_api(self, cloud):
        keys = []
        empty_keys = []
        kvs = self.get_all_kvs(cloud)
        with concurrent.futures.ThreadPoolExecutor() as executor:
            results = list(executor.map(lambda args: self.process_keys(*args), [(kv, cloud) for kv in kvs]))
            for k, ek in results:
                keys.extend(k)
                empty_keys.extend(ek)
        return keys, empty_keys

2. Why threads at all? (the problem being solved)

The job of this script is simple to describe:

Ask Azure Resource Graph: "give me every Key Vault, grouped by subscription"
↓
For each subscription, call the Key Vault API to list keys / secrets in every vault
↓
Collect everything into one big list and INSERT into SQL

The expensive part is the middle step. Every call like kv_client.keys.list(...) is a network request to Azure. A single request may take 200 ms–2 s. Your CPU does almost nothing during that time — it just waits for bytes to come back over the wire.

Sequential (no threads)

main thread
sub-1 (3s)
sub-2 (5s)
sub-3 (2s)
0s5s10s

Total = 3 + 5 + 2 = 10 seconds. Each subscription waits for the previous one to finish.

Concurrent (ThreadPoolExecutor — what this code does)

worker-1 → sub-1
3s
idle
worker-2 → sub-2
5s
worker-3 → sub-3
2s
idle
0s2.5s5s

Total = 5 seconds (the slowest one). All three waits overlap.

Core idea Threads don't make any single network call faster. They let you wait for many calls at the same time instead of one after another. Total time drops from "sum of all" to "the slowest one".

3. Threading basics — process, thread, and the GIL

TermBeginner explanation
ProcessYour running Python program. Has its own memory. When you run python script.py, that's one process.
ThreadAn independent "line of execution" inside the same process. All threads in a process share the same variables and memory. Cheap to create.
Main threadThe thread that starts when Python starts. In our code, this is the one running kv_keys_api.
Worker threadAn extra thread created by the pool to run your function. Here each worker runs one process_keys(...) call.
I/O-boundWork that spends most of its time waiting (network, disk, database). ✅ Threads help a lot. This script is I/O-bound.
CPU-boundWork that spends time calculating (math, image processing). ❌ Threads don't help in Python — use ProcessPoolExecutor.
GILGlobal Interpreter Lock. Only one thread runs Python bytecode at a time. But the GIL is released during network/disk waits — which is exactly why threading works beautifully here.
Rule of thumb Waiting for something outside Python (an API, a DB, a file)? → threads.
Crunching numbers inside Python? → processes.

What the GIL actually looks like

# Thread A                        # Thread B
build request dict     # holds GIL
send HTTP request      # ── releases GIL while waiting ──>  build request dict  (gets GIL)
   (waiting on socket)                                    send HTTP request
   (waiting on socket)                                       (waiting on socket)
response arrives       # <── reacquires GIL to parse
parse JSON

Both threads are "in flight" at the same time because the slow part (the socket wait) doesn't need the GIL.

4. What a ThreadPoolExecutor actually is

A thread pool is a reusable team of workers plus a queue of jobs.

📥 Job queue: (sub-1, cloud) · (sub-2, cloud) · (sub-3, cloud) · (sub-4, cloud) …
↓ workers pull jobs as they become free
👷 worker-1 → running sub-1
👷 worker-2 → running sub-2
👷 worker-3 → running sub-3
👷 worker-4 → running sub-4
↓ each finished job's return value is stored
📤 results (kept in submission order)

The with statement matters

with concurrent.futures.ThreadPoolExecutor() as executor:
    ...
# <- at this point executor.shutdown(wait=True) is called automatically.
#    Python blocks here until EVERY worker thread has finished, then
#    destroys the threads. No leaked threads.

How many workers do you get?

ThreadPoolExecutor() with no arguments uses the default:

max_workers = min(32, (os.cpu_count() or 1) + 4)

# 4-core machine   -> 8 workers
# 8-core machine   -> 12 workers
# 64-core machine  -> 32 workers (capped)
Important consequence If you have 50 subscriptions but only 12 workers, only 12 run at a time. The other 38 sit in the queue and start as workers free up. You can set it explicitly: ThreadPoolExecutor(max_workers=20).

5. Deep dive: executor.map() line by line

Here is the line, split into its parts:

results = list(executor.map(lambda args: self.process_keys(*args), [(kv, cloud) for kv in kvs]))
         └──┬──┘└───┬────┘└──────────────┬──────────────────┘  └──────────┬───────────────┘
            │       │                    │                                │
            │       │                    │                        ④ the LIST of inputs
            │       │            ③ the FUNCTION each worker runs
            │   ② schedule one call per input, on worker threads
        ① force the lazy iterator to actually run & collect everything

④ The input list

[(kv, cloud) for kv in kvs]

# kvs came from get_all_kvs(cloud) and looks like:
kvs = [
    {'subscriptionId': 'sub-1', 'tenantId': 't1', 'keyvaults': [...]},
    {'subscriptionId': 'sub-2', 'tenantId': 't1', 'keyvaults': [...]},
    {'subscriptionId': 'sub-3', 'tenantId': 't2', 'keyvaults': [...]},
]

# so the comprehension produces:
[
    ({'subscriptionId': 'sub-1', ...}, 'AzurePublicCloud'),   # item 0
    ({'subscriptionId': 'sub-2', ...}, 'AzurePublicCloud'),   # item 1
    ({'subscriptionId': 'sub-3', ...}, 'AzurePublicCloud'),   # item 2
]

Each element is a tuple of the arguments for one call. cloud is repeated in every tuple because it is the same for the whole batch.

③ The lambda and the *args trick

map only ever passes one item to the function. But process_keys needs two parameters. The lambda bridges that gap:

lambda args: self.process_keys(*args)
#     ▲                         ▲
#     │                         └── * unpacks the tuple into separate arguments
#     └── receives ONE item, e.g. args = (kv_dict, 'AzurePublicCloud')

# So this:
args = (kv_dict, 'AzurePublicCloud')
self.process_keys(*args)
# becomes exactly this:
self.process_keys(kv_dict, 'AzurePublicCloud')
Simpler alternative executor.map also accepts multiple iterables, so the lambda can be avoided entirely: 
results = list(executor.map(self.process_keys, kvs, itertools.repeat(cloud)))
Same behaviour, no lambda, no tuple building.

② + ① What map does internally

This is a simplified version of CPython's real implementation — it explains almost every question people have about map:

def map(self, fn, *iterables):
    fs = [self.submit(fn, *args) for args in zip(*iterables)]   # ① submit ALL jobs NOW, in order

    def result_iterator():
        for f in fs:            # ② walk the futures in SUBMISSION order
            yield f.result()      # ③ block until THIS future is done, then hand back its value

    return result_iterator()      # ④ a lazy generator — nothing is collected until you iterate it
ObservationWhy it matters
submit is called for every item immediatelyAll jobs are queued up-front; workers start right away. This is where concurrency begins.
It returns a generator, not a listThat's why the code wraps it in list(...). Without list() nothing would be collected and results would be consumed lazily.
It loops fs positionallyThis is the entire reason output order matches input order. See §7.
f.result() re-raises exceptionsIf a worker raised, the exception pops out here, in the main thread, and aborts the whole list().

6. Where the "multiple calls" come from

This is the part that confuses most people. There is one line of code, but it produces N function calls.

1
kvs is a list. Its length = number of unique (subscriptionId, tenantId) groups, because the KQL query ends with summarize keyvaults = make_list(bag, 10000) by subscriptionId, tenantId. If you have 40 subscriptions, len(kvs) == 40.
2
The comprehension turns those 40 items into 40 argument tuples.
3
map calls submit() once per tuple → 40 separate jobs, each a complete independent invocation of process_keys → keys_paginate.
4
Each of those 40 calls independently creates its own kv_client, loops its own vaults, builds its own local keys / empty_keys lists, and returns its own tuple.

Side-by-side: the same thing without threads

What the code does

with ThreadPoolExecutor() as ex:
    results = list(ex.map(
        lambda a: self.process_keys(*a),
        [(kv, cloud) for kv in kvs]
    ))

Identical output, no threads

results = []
for kv in kvs:
    results.append(self.process_keys(kv, cloud))

Same list, same order, same contents — the threaded version is just much faster because the waits overlap.

Two levels of looping — only the outer one is parallel

kv_keys_api
 └─ executor.map over kvs                # ← ✅ PARALLEL: one thread per subscription
     └─ keys_paginate(one subscription)
         └─ for kv in kvs['keyvaults']:        # ← ❌ sequential inside that thread
             └─ for kv_id in kv['id']:       # ← ❌ sequential inside that thread
                 └─ kv_client.keys.list(...)  # one network call at a time
Practical impact A subscription with 300 vaults still processes those 300 vaults one after another inside its single thread. If one subscription owns most of the vaults, it becomes the bottleneck and the pool sits mostly idle waiting for it.

7. Why order is preserved even if a thread finishes early

The common worry: "If sub-3 finishes first, won't it end up first in the list?" — No. Here's why.

The key object: a Future

When you submit() a job you immediately get back a Future — an empty box with a label. The box is created before any work happens, and it never moves.

fs[0] → sub-1
state: RUNNING
result: —
fs[1] → sub-2
state: RUNNING
result: —
fs[2] → sub-3
state: RUNNING
result: —

t = 0s — all three submitted, in input order. Boxes are empty.

fs[0] → sub-1
state: RUNNING
result: —
fs[1] → sub-2
state: RUNNING
result: —
fs[2] → sub-3
state: DONE ✔
result: (keys3, empty3)

t = 2s — sub-3 finished first. Its value is written into its own box, fs[2]. It does not jump to the front of anything.

fs[0] → sub-1
state: DONE ✔
result: (keys1, empty1)
fs[1] → sub-2
state: DONE ✔
result: (keys2, empty2)
fs[2] → sub-3
state: DONE ✔
result: (keys3, empty3)

t = 5s — everything done. The boxes are still in their original order.

The collector reads boxes by position, not by finish time

for f in fs:          # fs[0], then fs[1], then fs[2] — always this order
    yield f.result()  # waits ONLY if that specific box is still empty

Worked example — completion order ≠ output order

sub-1 (slow, 5s)
finishes 3rd
sub-2 (medium, 3s)
finishes 2nd
sub-3 (fast, 1s)
1st
0s2.5s5s
Collector stepWhat happensYielded
f = fs[0] (sub-1)Box empty → blocks until t=5s. Meanwhile sub-2 and sub-3 quietly fill their own boxes.sub-1 result
f = fs[1] (sub-2)Box already full since t=3s → returns instantly, no waiting.sub-2 result
f = fs[2] (sub-3)Box already full since t=1s → returns instantly.sub-3 result
# Completion order was:  sub-3, sub-2, sub-1
# But results is:
results = [
    (keys_sub1, empty_sub1),   # from fs[0]  ← input position 0
    (keys_sub2, empty_sub2),   # from fs[1]  ← input position 1
    (keys_sub3, empty_sub3),   # from fs[2]  ← input position 2
]
The one-sentence answer Finishing early only changes when a result becomes available — never where it is stored, because each result is written into its own pre-allocated Future at a fixed index in a list built at submission time.
Two useful corollaries 1. Total wall-clock time for list(map(...)) ≈ time of the slowest job (they overlap), not the first one.
2. If you'd rather process results the moment they arrive (completion order), use as_completed instead — see §12.

8. Returning two values & unpacking them

Python has no "two return values" — it has tuples

return keys, empty_keys
# The comma builds a tuple. This is IDENTICAL to:
return (keys, empty_keys)
# ONE object is returned. It happens to be a 2-element container.

Case A — direct call → unpack once

keys_pub, empty_keys_pub = self.kv_keys_api('AzurePublicCloud')

# right side evaluates to ONE tuple:  ([...keys...], [...empty...])
# two names on the left  => Python unpacks the tuple element-wise
#   keys_pub       = tuple[0]
#   empty_keys_pub = tuple[1]

Case B — via map → a list of tuples → unpack repeatedly

results = list(executor.map(...))

# results is a LIST, each element is one call's tuple:
results = [
    (keys1, empty1),
    (keys2, empty2),
    (keys3, empty3),
]

# You CANNOT do:  a, b = results        ← ValueError if len(results) != 2
# You unpack ONE ELEMENT AT A TIME:
for k, ek in results:
    keys.extend(k)
    empty_keys.extend(ek)

That loop is exactly equivalent to:

for item in results:
    k  = item[0]   # the keys list from that subscription
    ek = item[1]   # the empty-vault list from that subscription
    keys.extend(k)
    empty_keys.extend(ek)
So the pattern is the same in both cases Case A = "unpack a tuple, once". Case B = "unpack a tuple, N times (once per subscription)". map just produced N tuples instead of one.

append vs extend — why extend is used

keys = []
k = ['a', 'b']

keys.append(k)   # => [['a','b']]        nested list — WRONG here
keys.extend(k)   # => ['a', 'b']          flattened — what we want

Each thread returns a list of key records. extend pours those records into the one master list, keeping it flat so json.dumps produces a single flat array.

9. Full end-to-end trace of one run

Assume 3 subscriptions in AzurePublicCloud:

  • sub-1: 2 vaults → vaultA has 3 keys, vaultB has 0 keys
  • sub-2: 1 vault → vaultC has 5 keys
  • sub-3: 1 vault → vaultD has 0 keys
#### MAIN THREAD ####
kv_keys_api('AzurePublicCloud')
  keys = []            # master accumulators (main thread only)
  empty_keys = []

  kvs = get_all_kvs('AzurePublicCloud')
  # => [ {sub-1,...}, {sub-2,...}, {sub-3,...} ]     ← 3 items

  args = [ ({sub-1},cloud), ({sub-2},cloud), ({sub-3},cloud) ]

  executor.map(...)  → submits 3 jobs
      fs[0] = Future(sub-1)
      fs[1] = Future(sub-2)
      fs[2] = Future(sub-3)

#### WORKER THREADS (all running at the same time) ####

  [worker-1] process_keys({sub-1}, cloud) → keys_paginate
      kv_client = KeyVaultClient().get_kv_client('sub-1', cloud)   # its OWN client
      local keys=[] empty=[]
      vaultA → keys.list() → 3 keys  → append 3 dicts to LOCAL keys
      vaultB → keys.list() → []      → append 1 dict  to LOCAL empty
      return ([k1,k2,k3], [vaultB])            → written into fs[0]

  [worker-2] process_keys({sub-2}, cloud) → keys_paginate
      vaultC → 5 keys
      return ([k4,k5,k6,k7,k8], [])            → written into fs[1]

  [worker-3] process_keys({sub-3}, cloud) → keys_paginate
      vaultD → 0 keys
      return ([], [vaultD])                     → written into fs[2]

#### BACK ON MAIN THREAD ####
  results = list(...)   # reads fs[0], fs[1], fs[2] in that order
  results = [
      ([k1,k2,k3],        [vaultB]),
      ([k4,k5,k6,k7,k8],  []      ),
      ([],                [vaultD]),
  ]

  for k, ek in results:
      iter 1: k=[k1,k2,k3]         ek=[vaultB]   → keys=[k1,k2,k3]            empty=[vaultB]
      iter 2: k=[k4..k8]           ek=[]         → keys=[k1..k8]              empty=[vaultB]
      iter 3: k=[]                 ek=[vaultD]   → keys=[k1..k8]              empty=[vaultB,vaultD]

  return keys, empty_keys      # => ([8 key dicts], [2 empty-vault dicts])

#### CALLER: get_kv_objects ####
keys_pub, empty_keys_pub = # ← unpacks that single returned tuple
Why there's no race condition here Each worker builds its own local keys/empty_keys lists and returns them. Nothing is written to a shared list from inside a thread. Merging happens afterwards on the main thread only. This is the safest threading pattern — share nothing, return everything.

Script 3 — A miniature clone of kv_keys_api

# demo3_mini_kv.py  — same shape as the real code, fake data
import time, random, concurrent.futures

# pretend get_all_kvs() returned this
kvs = [
    {"subscriptionId": "sub-1", "tenantId": "t1", "vaults": ["vaultA", "vaultB"]},
    {"subscriptionId": "sub-2", "tenantId": "t1", "vaults": ["vaultC"]},
    {"subscriptionId": "sub-3", "tenantId": "t2", "vaults": ["vaultD", "vaultE", "vaultF"]},
]

def keys_paginate(sub, cloud):
    """One worker's job: returns (keys, empty_vaults) for ONE subscription."""
    keys, empty = [], []
    for vault in sub["vaults"]:
        time.sleep(random.uniform(0.3, 1.2))       # pretend keys.list() network call
        n = random.choice([0, 1, 2, 3])
        if n == 0:
            empty.append({"vault": vault, "sub": sub["subscriptionId"], "cloud": cloud})
        else:
            for i in range(n):
                keys.append({"name": f"{vault}-key{i}", "sub": sub["subscriptionId"]})
    print(f"  {sub['subscriptionId']} done: {len(keys)} keys, {len(empty)} empty vaults")
    return keys, empty                              # ← ONE tuple

def kv_keys_api(cloud):
    keys, empty_keys = [], []
    with concurrent.futures.ThreadPoolExecutor() as ex:
        results = list(ex.map(lambda a: keys_paginate(*a), [(kv, cloud) for kv in kvs]))
        print("\nresults is a LIST of TUPLES:")
        for i, r in enumerate(results):
            print(f"  results[{i}] = (list of {len(r[0])} keys, list of {len(r[1])} empty)")
        for k, ek in results:                     # unpack one tuple per iteration
            keys.extend(k)
            empty_keys.extend(ek)
    return keys, empty_keys

k, ek = kv_keys_api("AzurePublicCloud")          # unpack the single returned tuple
print(f"\nTOTAL: {len(k)} keys, {len(ek)} empty vaults")

12. map vs submit vs as_completed

ApproachResult orderError behaviourBest for
executor.map Input order First exception aborts the whole list() Same function over a uniform list, order matters. ✅ what this code uses
submit + as_completed Completion order You handle each future's error individually Streaming/progress reporting, resilience to per-job failures
submit + index dict You choose Per-job, and you know which input failed Best of both — recommended for production

Same work, three ways

# A) map — ordered, simple, all-or-nothing
with ThreadPoolExecutor() as ex:
    results = list(ex.map(lambda a: process_keys(*a), [(kv, cloud) for kv in kvs]))


# B) as_completed — react as soon as each one lands
with ThreadPoolExecutor() as ex:
    futures = [ex.submit(process_keys, kv, cloud) for kv in kvs]
    for fut in concurrent.futures.as_completed(futures):   # finish order!
        k, ek = fut.result()
        keys.extend(k); empty_keys.extend(ek)


# C) submit + mapping — know WHICH input failed, keep going  ← recommended
with ThreadPoolExecutor(max_workers=16) as ex:
    future_to_sub = {ex.submit(process_keys, kv, cloud): kv['subscriptionId'] for kv in kvs}
    for fut in concurrent.futures.as_completed(future_to_sub):
        sub = future_to_sub[fut]
        try:
            k, ek = fut.result()
            keys.extend(k); empty_keys.extend(ek)
        except Exception as e:
            logging.error(f"subscription {sub} failed: {e}")   # others still complete

13. Pitfalls in this code & how to improve

⚠️ 1. Uneven work — parallelism is only per subscription

One subscription with 500 vaults runs all 500 sequentially in a single thread while other workers sit idle. Fix: flatten to one job per vault:

# build a flat job list: one entry per vault
jobs = [
    (sub['subscriptionId'], sub['tenantId'], grp['resource_group'], vault_id, cloud)
    for sub in kvs
    for grp in sub['keyvaults']
    for vault_id in grp['id']
]
# now every vault is an independent unit of work -> far better load balancing
# (cache the kv_client per subscription so you don't rebuild it per vault)

⚠️ 2. Unbounded worker count / API throttling

Default workers scale with CPU count and each hammers the Azure API. Azure Resource Manager throttles at ~1200 reads/hour per subscription. Fix: pin it explicitly and tune.

with ThreadPoolExecutor(max_workers=16) as executor:   # deliberate, not accidental

⚠️ 3. One failure kills the batch

keys_paginate catches errors per-vault, which is good. But if anything escapes (e.g. a bug in the accumulation code), list(executor.map(...)) re-raises it on the main thread and you lose all results collected so far. Fix: pattern C in §12.

⚠️ 4. No timeout — one hung call blocks everything

f.result() waits forever by default. If one Azure call hangs, list(...) never returns and the Function times out. Fix:

results = list(ex.map(fn, args, timeout=600))   # raises TimeoutError instead of hanging

⚠️ 5. Memory — everything is held in RAM at once

All keys from all subscriptions across both clouds are accumulated, then json.dumps'd into a single string for one SQL insert. With tens of thousands of records this can spike memory. Fix: batch the inserts.

✅ What this code does right

  • Share nothing — each thread builds local lists and returns them; merging happens on the main thread. No locks needed, no race conditions.
  • One client per thread — KeyVaultClient() is created inside the worker, avoiding shared-client thread-safety questions.
  • Per-vault try/except — a single bad vault doesn't kill the subscription's whole scan.
  • Context manager — with ThreadPoolExecutor() guarantees clean shutdown.
The golden rule of threading Never mutate a shared list/dict from multiple threads without a Lock. This code avoids the problem entirely by returning values instead of writing to shared state — copy that habit.

14. Cheat sheet

import concurrent.futures, itertools

# ── create a pool (always use `with`) ─────────────────────────────
with concurrent.futures.ThreadPoolExecutor(max_workers=16) as ex:

    # one arg per call
    results = list(ex.map(fn, items))

    # two args, constant second arg
    results = list(ex.map(fn, items, itertools.repeat(cloud)))

    # two args via tuples + lambda  (what this codebase uses)
    results = list(ex.map(lambda a: fn(*a), [(i, cloud) for i in items]))

    # fire one job, get a Future back
    fut = ex.submit(fn, arg1, arg2)
    value = fut.result()               # blocks; re-raises worker exceptions
    value = fut.result(timeout=30)     # or TimeoutError

    # process in completion order
    futs = [ex.submit(fn, i) for i in items]
    for f in concurrent.futures.as_completed(futs):
        ...
# leaving the `with` block waits for ALL workers, then closes them
QuestionAnswer
Does map preserve order?Yes — always input order, regardless of finish times.
How many calls does map make?Exactly len(iterable) — one per item.
Why list(...) around map?map returns a lazy generator; list() forces it to run and collect.
Why the lambda?map passes one item; *args splits that tuple into two parameters.
Does return a, b return two things?No — one tuple (a, b). Unpacking on assignment splits it.
How long does the whole thing take?≈ the slowest job (if workers ≥ jobs), not the sum.
Threads or processes?Waiting on network/disk → threads. Heavy CPU math → processes.
Is this code thread-safe?Yes — each worker owns its data and returns it; no shared mutation.
What if a worker raises?list(map(...)) re-raises it in the main thread and abandons the rest.
